\section{Motivation}

\begin{frame}{Why Distributional RL?}
  \begin{itemize}
    \item Classical RL tracks only the expected return: $Q(s,a) = \E[\sum_t \gamma^t r_t]$.
    \item In many tasks, \textbf{uncertainty and tail behavior} matter.
    \item Distributional RL models the full random return, not just its mean.
    \item This often improves stability and empirical performance.
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Takeaway:} Value distributions contain useful learning signal.
\end{frame}

\begin{frame}{Quick RL Recap (for context)}
  We assume an MDP with state $s$, action $a$, reward $r$, discount $\gamma \in [0,1)$.
  \begin{itemize}
    \item Policy: $\pi(a\mid s)$
    \item Classical Bellman optimality equation:
    \[
      Q^*(s,a) = \E\left[r + \gamma \max_{a'} Q^*(s',a') \mid s,a\right]
    \]
    \item DQN approximates $Q^*(s,a)$ with a neural network.
  \end{itemize}
\end{frame}

\section{Core Idea}

\begin{frame}{Return Distribution Instead of Mean}
  Define random return under policy $\pi$:
  \[
    G_t = \sum_{k=0}^{\infty} \gamma^k R_{t+k+1}
  \]
  Distributional RL models:
  \[
    Z^{\pi}(s,a) \stackrel{D}{=} G_t \mid (S_t=s, A_t=a)
  \]
  where $\stackrel{D}{=}$ denotes equality in distribution.
  \vspace{0.3cm}

  Expected value is recovered by:
  \[
    Q^{\pi}(s,a) = \E[Z^{\pi}(s,a)]
  \]
\end{frame}

\begin{frame}{Distributional Bellman Operator}
  The distributional Bellman update is:
  \[
    (\cT^{\pi} Z)(s,a) \stackrel{D}{=} R(s,a) + \gamma Z(S',A')
  \]
  with $A' \sim \pi(\cdot\mid S')$.

  \vspace{0.2cm}
  Optimal-control version:
  \[
    (\cT Z)(s,a) \stackrel{D}{=} R(s,a) + \gamma Z\bigl(S', a^*(S')\bigr)
  \]
  where $a^*(S') = \arg\max_{a'} \E[Z(S',a')]$.

  \vspace{0.3cm}
  \textbf{Important:} Contraction properties depend on the metric (e.g. Wasserstein/Cramer).
\end{frame}

\section{Main Algorithms}

\begin{frame}{C51 (Categorical DQN)}
  Idea: represent each $Z(s,a)$ on fixed support atoms $\{z_i\}_{i=1}^N$.
  \begin{itemize}
    \item Network outputs probabilities over atoms per action.
    \item Bellman target distribution is projected back to fixed support.
    \item Training minimizes cross-entropy between predicted and projected target.
  \end{itemize}

  \vspace{0.2cm}
  \textbf{Pros:} Simple, first major success.

  \textbf{Limitations:} Fixed support can be restrictive.
\end{frame}

\begin{frame}{QR-DQN and IQN}
  \textbf{QR-DQN (Quantile Regression DQN)}
  \begin{itemize}
    \item Learns quantile locations directly.
    \item Uses quantile Huber loss.
  \end{itemize}

  \vspace{0.2cm}
  \textbf{IQN (Implicit Quantile Networks)}
  \begin{itemize}
    \item Learns a continuous quantile function $F_Z^{-1}(\tau)$.
    \item Samples $\tau \sim \mathcal{U}(0,1)$ during training/inference.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Rule of thumb:} QR-DQN and IQN are often stronger baselines than C51.
\end{frame}

\section{Practical View}

\begin{frame}{Why It Works in Practice (High-level)}
  \begin{itemize}
    \item Richer target than scalar TD target.
    \item Better gradient signal from multiple quantiles/atoms.
    \item Improved robustness in stochastic environments.
    \item Natural bridge to risk-sensitive control.
  \end{itemize}
\end{frame}

\begin{frame}{Risk-sensitive Decision Making}
  Once we estimate $Z(s,a)$, we can optimize objectives beyond expectation:
  \begin{itemize}
    \item Mean (risk-neutral): $\E[Z]$
    \item CVaR / lower quantiles (risk-averse)
    \item Upper quantiles (risk-seeking)
  \end{itemize}

  \vspace{0.3cm}
  Example objective:
  \[
    a^*(s) = \arg\max_a \operatorname{CVaR}_{\alpha}(Z(s,a))
  \]
\end{frame}
